   117→
   118→  function handleDrop(e: React.DragEvent<HTMLButtonElement>) {
   119→    e.preventDefault();

   121→    const f = e.dataTransfer.files?.[0];
   122→    if (f) onFile(f);

   127→      <input
   128→        ref={inputRef}
   129→        type="file"
   130→        accept={ACCEPTED_IMAGE_TYPES.join(",")}

   132→        onChange={(e: ChangeEvent<HTMLInputElement>) => {
   133→          const f = e.target.files?.[0];

   135→          e.target.value = "";

   168→                src={preview}
   169→                alt=""
   170→                className="h-12 w-12 rounded border border-[#D6B77A]/30 object-cover"

   172→            )}
   173→            <div className="min-w-0 flex-1">
   174→              <div className="flex items-center gap-2">
   175→                <CheckCircle2 className="h-4 w-4 shrink-0 text-green-400" />

   177→              </div>
   178→              <span className="text-xs text-ivory/50">
   179→                Uploaded — click to replace
   180→              </span>
   181→            </div>
   182→          </div>
   183→        ) : (
   184→          <div className="flex w-full items-center gap-3">
